 1. Using OPENROWSET (Recommended for Ad-Hoc or Direct Querying):
This is the simplest way to read data from Parquet files without creating persistent external objects.

SQL

SELECT TOP 100 *
FROM OPENROWSET(
    BULK 'https://yourstorageaccount.blob.core.windows.net/yourcontainer/your_data.parquet',
    FORMAT = 'PARQUET'
) AS [result];
To ingest into a table: You can use INSERT INTO ... SELECT * FROM OPENROWSET(...).
SQL

CREATE TABLE YourTable (
    Column1 INT,
    Column2 VARCHAR(255),
    -- ... other columns

INSERT INTO YourTable

    Column1,
    Column2
FROM OPENROWSET(
    BULK 'https://yourstorageaccount.blob.core.windows.net/yourcontainer/your_data.parquet',
    FORMAT = 'PARQUET'
) AS [result];

-- Step 2: Create an External Data Source
-- This links your Dedicated SQL Pool to the ADLS Gen2 location where your Parquet files are.
-- Ensure the IDENTITY used here has permissions to access the ADLS Gen2 path.
-- 'MANAGED IDENTITY' is recommended as it uses the Synapse Workspace's System-Assigned Managed Identity.

IF NOT EXISTS (SELECT * FROM sys.external_data_sources WHERE name = 'ADLSGen2DataSource')
BEGIN
    CREATE EXTERNAL DATA SOURCE ADLSGen2DataSource
    WITH (
        LOCATION = 'abfss://<YourADLSGen2ContainerName>@<YourADLSGen2AccountName>.dfs.core.windows.net/',
        -- Using Managed Identity (recommended for Synapse)
        IDENTITY = 'Managed Identity'
        -- Or if using a Database Scoped Credential created in Step 1:
        -- CREDENTIAL = [<YourCredentialName>]
    );
END
GO

-- Step 3: Create an External File Format for Parquet
-- This defines how the SQL Pool should interpret the Parquet files.

IF NOT EXISTS (SELECT * FROM sys.external_file_formats WHERE name = 'ParquetFileFormat')
BEGIN
    CREATE EXTERNAL FILE FORMAT ParquetFileFormat
    WITH (
        FORMAT_TYPE = PARQUET,
        DATA_COMPRESSION = 'org.apache.hadoop.io.compress.SnappyCodec' -- Common compression for Parquet
        -- For uncompressed Parquet, use DATA_COMPRESSION = ''
    );
END
GO

-- Step 4: Create the Target Table in Dedicated SQL Pool
-- This is the internal table where your data will be loaded.
-- The schema (column names and data types) must match your Parquet file's schema.
-- Adjust column definitions (names, data types, nullability) as per your Parquet file.
-- Example: Assuming your Parquet has columns like ProductID (int), ProductName (varchar), Price (decimal), etc.

IF OBJECT_ID('dbo.YourTargetTableName') IS NOT NULL
BEGIN
    DROP TABLE dbo.YourTargetTableName; -- Drop table if it exists (for re-running example)
END

CREATE TABLE dbo.YourTargetTableName
(
    -- Example Columns - Adjust these to match your Parquet file's schema
    ProductID   INT,
    ProductName VARCHAR(255),
    Category    VARCHAR(100),
    Price       DECIMAL(10, 2),
    SaleDate    DATE,
    Quantity    INT
)
WITH
(
    DISTRIBUTION = ROUND_ROBIN, -- Or HASH(<YourDistributionColumn>) for performance, e.g., HASH(ProductID)
    HEAP -- Or CLUSTERED COLUMNSTORE INDEX for large tables and analytical queries
);
GO

-- Step 5: Ingest data using COPY INTO statement
-- This command reads the Parquet file(s) from your ADLS Gen2 and loads them into the target table.
-- You can specify a single file or a folder path (which will ingest all Parquet files in that folder).

COPY INTO dbo.YourTargetTableName
FROM '<PathToYourParquetFileOrFolderInContainer>' -- e.g., 'sales/2024/product_data.parquet' or 'sales/2024/'
WITH (
    FILE_TYPE = 'PARQUET',
    -- If your Parquet files are within subfolders of the external data source's location,
    -- this path should be relative to the external data source's LOCATION.
    -- Example: If EXTERNAL DATA SOURCE LOCATION is 'abfss://container@account.dfs.core.windows.net/'
    -- and your file is 'abfss://container@account.dfs.core.windows.net/sales/2024/product_data.parquet',
    -- then FROM path is 'sales/2024/product_data.parquet'
    
    -- Specify the external data source and file format created previously
    EXTERNAL_DATA_SOURCE = 'ADLSGen2DataSource',
    EXTERNAL_FILE_FORMAT = 'ParquetFileFormat',

    -- Optional: ERRORFILE for logging bad rows
    -- ERRORFILE = 'https://<YourADLSGen2AccountName>.dfs.core.windows.net/<ErrorContainerName>/errors/',
    -- ERRORFILE_CREDENTIAL = [<YourCredentialNameForErrorFile>] -- If different credential for error file path
    
    -- Optional: If you need to skip header, specify ROWTERMINATOR.
    -- For Parquet, skipping header is usually not necessary as schema is embedded.
    -- ROWTERMINATOR='0x0A' -- For CSV files
    
    -- Optional: MAXERRORS to allow a certain number of errors before failing
    -- MAXERRORS = 0 -- Default is 0, means no errors allowed
    
    -- Optional: Specify the collation for string columns if needed
    -- CODEPAGE = '65001' -- For UTF-8 encoding
);
GO

-- Step 6: Verify the data (Optional)
SELECT TOP 100 *
FROM dbo.YourTargetTableName;
GO

WITH (
    LOCATION = '/silver/sales_transformed/',
    DATA_SOURCE = MyDataLake,

Scenario: Replicate a table Sales.Orders from On-Premise SQL Server (Publisher/Distributor) to an Azure SQL Database (Subscriber). We'll replicate only orders from 2023 and only the OrderID, OrderDate, and TotalAmount columns.

Step 1: On-Premise SQL Server - Configure Distributor (T-SQL)
Run this on your on-premise SQL Server instance that will act as the Distributor.

SQL

-- Enable the Distributor (can be the same server as the Publisher)
USE master;
EXEC sp_adddistributor
    @distributor = N'$(YourOnPremSQLServerName)', -- Replace with your on-prem SQL Server name
    @password = N'YourDistributorPassword!'; -- Choose a strong password

EXEC sp_adddistributiondb
    @database = N'distribution',
    @data_folder = N'C:\Program Files\Microsoft SQL Server\MSSQL16.MSSQLSERVER\MSSQL\Data', -- Adjust path as needed
    @log_folder = N'C:\Program Files\Microsoft SQL Server\MSSQL16.MSSQLSERVER\MSSQL\Data', -- Adjust path as needed
    @security_mode = 1; -- Windows Authentication for connection to Distributor

-- Add the Publisher to the Distributor
EXEC sp_adddistpublisher
    @publisher = N'$(YourOnPremSQLServerName)', -- Replace with your on-prem SQL Server name
    @distribution_db = N'distribution',
    @security_mode = 1, -- Windows Authentication for connection from Publisher to Distributor
    @working_directory = N'\\YourAzureFileShare.file.core.windows.net\replication_snapshot_share'; -- Path to your Azure Files share
    -- Ensure the Azure Files share is mapped as a network drive on the on-prem server
    -- and the SQL Server Agent account has access.

6. Export Data using CREATE EXTERNAL TABLE AS SELECT (CETAS):

SQL

-- This statement creates the external table and exports the data in one go.
-- It will create new Parquet files in the specified location.
CREATE EXTERNAL TABLE dbo.ExportedOrdersParquet
WITH (
    LOCATION = '/exports/orders/', -- Directory within your ADLS container
    DATA_SOURCE = MyADLSDataSource,
    FILE_FORMAT = ParquetFormat
)
AS
SELECT
    OrderID,
    CustomerID,
    OrderDate,
    TotalAmount,
    OrderStatus
FROM
    dbo.MyOrders
WHERE
    OrderDate >= '2023-01-01'; -- Example filter
GO
Alternatively, Export using INSERT INTO <external_table> SELECT:

First, define the external table schema:

SQL

CREATE EXTERNAL TABLE dbo.MyOrdersExternal (
    OrderID INT,
    CustomerID INT,
    OrderDate DATE,
    TotalAmount DECIMAL(18, 2),
    OrderStatus NVARCHAR(50)
)
WITH (
    LOCATION = '/exports/moreorders/', -- Another directory for these exports
    DATA_SOURCE = MyADLSDataSource,
    FILE_FORMAT = ParquetFormat
);
GO

-- Then, insert data into it
INSERT INTO dbo.MyOrdersExternal
SELECT
    OrderID,
    CustomerID,
    OrderDate,
    TotalAmount,
    OrderStatus
FROM
    dbo.MyOrders
WHERE
    OrderDate < '2023-01-01';
GO
