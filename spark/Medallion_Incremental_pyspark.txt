# Databricks Notebook: PySpark Medallion Architecture with Incremental Updates

# This notebook demonstrates how to implement a Medallion Architecture (Bronze, Silver, Gold layers)
# using PySpark and Delta Lake tables in a Databricks environment, focusing on incremental updates.

# --- Define ETL Functions for Each Layer ---

def run_bronze_layer(new_raw_data_df, mode="append"):
    """
    Ingests new raw data into the Bronze layer.
    """
    print(f"\n--- Bronze Layer: Ingesting New Raw Data ({mode} mode) ---")
    
    # Write raw data to Bronze Delta table
    # Schema evolution enabled for raw layer robustness
    new_raw_data_df.write \
        .format("delta") \
        .mode(mode) \
        .option("path", bronze_path) \
        .option("mergeSchema", "true") \
        .saveAsTable("raw_customer_orders")
    
    print(f"New raw data ingested into Bronze layer at: {bronze_path}")
    print("Current Bronze Table Schema:")
    spark.sql("DESCRIBE raw_customer_orders").show()
    print("Current Bronze Table Sample Data (latest 10 rows):")
    spark.sql("SELECT * FROM raw_customer_orders ORDER BY TimestampUTC_raw DESC LIMIT 10").show()

def run_silver_layer():
    """
    Cleanses and conforms data from Bronze, upserting into Silver.
    """
    print("\n--- Silver Layer: Cleaning & Conforming Data (Incremental Merge) ---")

    # Read all data from Bronze (can optimize to read only new changes if using CDF)
    # For simplicity, we read the full Bronze table here, and MERGE handles existing records.
    bronze_df = spark.read.format("delta").load(bronze_path)

    # Apply transformations for the Silver layer (similar to previous example)
    silver_source_df = bronze_df.select(
        col("OrderID_raw").cast(IntegerType()).alias("order_id"),
        coalesce(col("CustomerID_raw"), lit("UNKNOWN")).alias("customer_id"),
        col("Item_raw").alias("item_name"),
        col("Quantity_raw").cast(IntegerType()).alias("quantity"),
        to_date(col("OrderDate_raw"), "yyyy-MM-dd").alias("order_date"),
        col("TimestampUTC_raw").alias("raw_timestamp_utc"),
        current_timestamp().alias("processing_timestamp") # Add processing timestamp
    )

    # Apply data quality rules (e.g., filter out invalid quantities)
    silver_source_df = silver_source_df.filter(col("quantity") > 0) \
                                       .filter(col("order_id").isNotNull()) # Ensure order_id is not null after cast

    # Check if Silver table exists. If not, create it with overwrite. Else, perform MERGE.
    if not DeltaTable.isDeltaTable(spark, silver_path):
        print("Silver table does not exist. Performing initial load (overwrite).")
        silver_source_df.write \
            .format("delta") \
            .mode("overwrite") \
            .option("path", silver_path) \
            .saveAsTable("conformed_customer_orders")
    else:
        print("Silver table exists. Performing incremental merge.")
        silver_delta_table = DeltaTable.forPath(spark, silver_path)
        
        # Merge new/updated data from source_df into the target_silver_table
        # This handles both updates to existing orders and inserts for new orders
        silver_delta_table.alias("target") \
            .merge(
                silver_source_df.alias("source"),
                "target.order_id = source.order_id"
            ) \
            .whenMatchedUpdate(set = { # Update existing records
                "customer_id": "source.customer_id",
                "item_name": "source.item_name",
                "quantity": "source.quantity",
                "order_date": "source.order_date",
                "raw_timestamp_utc": "source.raw_timestamp_utc",
                "processing_timestamp": "source.processing_timestamp"
            }) \
            .whenNotMatchedInsert(values = { # Insert new records
                "order_id": "source.order_id",
                "customer_id": "source.customer_id",
                "item_name": "source.item_name",
                "quantity": "source.quantity",
                "order_date": "source.order_date",
                "raw_timestamp_utc": "source.raw_timestamp_utc",
                "processing_timestamp": "source.processing_timestamp"
            }) \
            .execute()

    print(f"Cleaned data incrementally updated in Silver layer at: {silver_path}")
    print("Current Silver Table Schema:")
    spark.sql("DESCRIBE conformed_customer_orders").show()
    print("Current Silver Table Sample Data:")
    spark.sql("SELECT * FROM conformed_customer_orders ORDER BY order_id LIMIT 10").show()

def run_gold_layer():
    """
    Aggregates data from Silver, upserting into Gold.
    """
    print("\n--- Gold Layer: Curating Data for Business Intelligence (Incremental Merge) ---")

    # Read all data from Silver layer
    silver_conformed_df = spark.read.format("delta").load(silver_path)

    # Aggregate monthly sales by item
    gold_source_df = silver_conformed_df.groupBy(
        year(col("order_date")).alias("sales_year"),
        month(col("order_date")).alias("sales_month"),
        col("item_name")
    ).agg(
        sum("quantity").alias("total_quantity_sold")
    )

    # Check if Gold table exists. If not, create it with overwrite. Else, perform MERGE.
    if not DeltaTable.isDeltaTable(spark, gold_path):
        print("Gold table does not exist. Performing initial load (overwrite).")
        gold_source_df.write \
            .format("delta") \
            .mode("overwrite") \
            .option("path", gold_path) \
            .saveAsTable("monthly_item_sales")
    else:
        print("Gold table exists. Performing incremental merge.")
        gold_delta_table = DeltaTable.forPath(spark, gold_path)
        
        # Merge new/updated aggregated data into the target Gold table
        gold_delta_table.alias("target") \
            .merge(
                gold_source_df.alias("source"),
                "target.sales_year = source.sales_year AND target.sales_month = source.sales_month AND target.item_name = source.item_name"
            ) \
            .whenMatchedUpdate(set = { # Update existing aggregates
                "total_quantity_sold": "source.total_quantity_sold"
            }) \
            .whenNotMatchedInsert(values = { # Insert new aggregates
                "sales_year": "source.sales_year",
                "sales_month": "source.sales_month",
                "item_name": "source.item_name",
                "total_quantity_sold": "source.total_quantity_sold"
            }) \
            .execute()

    print(f"Aggregated data incrementally updated in Gold layer at: {gold_path}")
    print("Current Gold Table Schema:")
    spark.sql("DESCRIBE monthly_item_sales").show()
    print("Current Gold Table Sample Data:")
    spark.sql("SELECT * FROM monthly_item_sales ORDER BY sales_year, sales_month, item_name").show()
