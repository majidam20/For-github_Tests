# --- Layer Processing Functions (using PySpark SQL) ---

# Bronze Layer Functions
def run_bronze_orders(new_raw_df, mode="append"):
    """Appends new raw order data to the Bronze layer Delta table."""
    print(f"\n--- Bronze Layer: Ingesting Raw Order Data ({mode} mode) ---")
    
    # Create a temporary view from the incoming DataFrame
    new_raw_df.createOrReplaceTempView("new_raw_orders_staging")

    # Write to Bronze Delta table (or create if not exists)
    # Using mergeSchema for robustness against source schema changes in Bronze
    spark.sql(f"""
        CREATE TABLE IF NOT EXISTS raw_customer_orders
        USING DELTA
        LOCATION '{bronze_orders_path}'
        TBLPROPERTIES ('delta.mergeSchema' = 'true');
    """)
    
    spark.sql(f"""
        INSERT INTO raw_customer_orders
        SELECT * FROM new_raw_orders_staging;
    """)

    print(f"Raw order data ingested into Bronze layer at: {bronze_orders_path}")
    print("Current Bronze Table Sample Data (latest 5 rows):")
    spark.sql("SELECT * FROM raw_customer_orders ORDER BY TimestampUTC_raw DESC LIMIT 5").show()

def run_bronze_customer_updates(new_raw_df, mode="append"):

# Silver Layer Functions
def run_silver_orders():
    """
    Cleanses and conforms order data from Bronze, upserting into Silver.
    Uses MERGE INTO for incremental updates.
    """
    print("\n--- Silver Layer: Processing Conformed Orders (Incremental Merge) ---")

    # Create a temporary view of cleaned/conformed orders from Bronze
    spark.sql(f"""
        CREATE OR REPLACE TEMPORARY VIEW conformed_orders_staging AS
        SELECT
            CAST(OrderID_raw AS INT) AS order_id,
            COALESCE(CustomerID_raw, 'UNKNOWN') AS customer_id,
            Item_raw AS item_name,
            CAST(Quantity_raw AS INT) AS quantity,
            TO_DATE(OrderDate_raw, 'yyyy-MM-dd') AS order_date, -- Handles 'yyyy/MM/dd' if TO_DATE supports it implicitly or use FORMAT_STRING
            TO_TIMESTAMP(TimestampUTC_raw, "yyyy-MM-dd HH:mm:ss'Z'") AS order_timestamp_utc,
            CURRENT_TIMESTAMP() AS processing_timestamp
        FROM raw_customer_orders
        WHERE CAST(Quantity_raw AS INT) > 0 AND OrderID_raw IS NOT NULL AND TRIM(OrderID_raw) <> '';
    """)

    # Check if Silver table exists. If not, create it with overwrite. Else, perform MERGE.
    if not DeltaTable.isDeltaTable(spark, silver_orders_path):
        print("Conformed Orders table does not exist. Performing initial load (overwrite).")
        spark.sql(f"""
            CREATE TABLE conformed_customer_orders
            USING DELTA
            LOCATION '{silver_orders_path}'
            AS SELECT * FROM conformed_orders_staging;
        """)
    else:
        print("Conformed Orders table exists. Performing incremental merge.")
        spark.sql(f"""
            MERGE INTO conformed_customer_orders AS target
            USING conformed_orders_staging AS source
            ON target.order_id = source.order_id
            WHEN MATCHED THEN
                UPDATE SET
                    target.customer_id = source.customer_id,
                    target.item_name = source.item_name,
                    target.quantity = source.quantity,
                    target.order_date = source.order_date,
                    target.order_timestamp_utc = source.order_timestamp_utc,
                    target.processing_timestamp = source.processing_timestamp
            WHEN NOT MATCHED THEN
                INSERT (order_id, customer_id, item_name, quantity, order_date, order_timestamp_utc, processing_timestamp)
                VALUES (source.order_id, source.customer_id, source.item_name, source.quantity, source.order_date, source.order_timestamp_utc, source.processing_timestamp);
        """)
    print(f"Conformed order data incrementally updated in Silver layer at: {silver_orders_path}")
    print("Current Conformed Orders Sample Data (latest 5 rows):")
    spark.sql("SELECT * FROM conformed_customer_orders ORDER BY order_id LIMIT 5").show()


def run_silver_dim_customer():
    """
    Implements SCD Type 2 for the Customer Dimension table.
    Reads from Bronze customer updates and uses MERGE INTO for SCD Type 2 logic.
    """
    print("\n--- Silver Layer: Processing Customer Dimension (SCD Type 2) ---")

    # Create a temporary view of the latest customer updates from Bronze
    # Using QUALIFY to get the latest update for each customer in the current batch
    spark.sql(f"""
        CREATE OR REPLACE TEMPORARY VIEW latest_customer_updates_view AS
        SELECT
            CustomerID_raw AS customer_id,
            CustomerName_raw AS customer_name,
            CustomerEmail_raw AS customer_email,
            TO_TIMESTAMP(UpdateTimestamp_raw, "yyyy-MM-dd HH:mm:ss'Z'") AS effective_timestamp
        FROM raw_customer_updates
        QUALIFY ROW_NUMBER() OVER (PARTITION BY CustomerID_raw ORDER BY TO_TIMESTAMP(UpdateTimestamp_raw, "yyyy-MM-dd HH:mm:ss'Z'") DESC) = 1;
    """)

    # Check if dim_customer table exists. If not, create it with initial load.
    if not DeltaTable.isDeltaTable(spark, silver_dim_customer_path):
        print("dim_customer table does not exist. Performing initial load (overwrite).")
        spark.sql(f"""
            CREATE TABLE dim_customer
            USING DELTA
            LOCATION '{silver_dim_customer_path}'
            AS SELECT
                customer_id,
                customer_name,
                customer_email,
                CAST(effective_timestamp AS DATE) AS start_date,
                CAST('9999-12-31' AS DATE) AS end_date,
                TRUE AS is_current
            FROM latest_customer_updates_view;
        """)
    else:
        print("dim_customer table exists. Performing SCD Type 2 merge.")
        
        # Step 1: Set is_current = FALSE and update end_date for changed records
        spark.sql(f"""
            MERGE INTO dim_customer AS target
            USING latest_customer_updates_view AS source
            ON target.customer_id = source.customer_id
               AND target.is_current = TRUE
               AND (target.customer_name <> source.customer_name OR target.customer_email <> source.customer_email)
            WHEN MATCHED THEN
                UPDATE SET target.end_date = CURRENT_DATE(), target.is_current = FALSE;
        """)

        # Step 2: Insert new records (both entirely new customers and new versions of updated customers)
        # This MERGE is designed to only insert if there is no *current and identical* record.
        # It handles new customers and new versions of existing customers (which were closed in Step 1).
        spark.sql(f"""
            MERGE INTO dim_customer AS target
            USING latest_customer_updates_view AS source
            ON target.customer_id = source.customer_id
               AND target.is_current = TRUE
               AND (target.customer_name = source.customer_name AND target.customer_email = source.customer_email)
            WHEN NOT MATCHED THEN
                INSERT (customer_id, customer_name, customer_email, start_date, end_date, is_current)
                VALUES (source.customer_id, source.customer_name, source.customer_email, CURRENT_DATE(), CAST('9999-12-31' AS DATE), TRUE);
        """)
    print(f"Customer dimension incrementally updated (SCD Type 2) in Silver layer at: {silver_dim_customer_path}")
    print("Current dim_customer Table Sample Data:")
    spark.sql("SELECT * FROM dim_customer ORDER BY customer_id, start_date DESC").show()


# Gold Layer Function
def run_gold_layer():
    """
    Aggregates data from Silver, upserting into Gold.
    Uses MERGE INTO for incremental updates of aggregates.
    """
    print("\n--- Gold Layer: Curating Data for Business Intelligence (Incremental Merge) ---")

    # Create a temporary view of aggregated data from Silver
    spark.sql(f"""
        CREATE OR REPLACE TEMPORARY VIEW monthly_item_sales_staging AS
        SELECT
            YEAR(order_date) AS sales_year,
            MONTH(order_date) AS sales_month,
            item_name,
            SUM(quantity) AS total_quantity_sold
        FROM conformed_customer_orders
        GROUP BY 1, 2, 3;
    """)

    # Check if Gold table exists. If not, create it with overwrite. Else, perform MERGE.
    if not DeltaTable.isDeltaTable(spark, gold_sales_path):
        print("Monthly Item Sales table does not exist. Performing initial load (overwrite).")
        spark.sql(f"""
            CREATE TABLE monthly_item_sales
            USING DELTA
            LOCATION '{gold_sales_path}'
            AS SELECT * FROM monthly_item_sales_staging;
        """)
    else:
        print("Monthly Item Sales table exists. Performing incremental merge.")
        spark.sql(f"""
            MERGE INTO monthly_item_sales AS target
            USING monthly_item_sales_staging AS source
            ON target.sales_year = source.sales_year
               AND target.sales_month = source.sales_month
               AND target.item_name = source.item_name
            WHEN MATCHED THEN
                UPDATE SET target.total_quantity_sold = source.total_quantity_sold
            WHEN NOT MATCHED THEN
                INSERT (sales_year, sales_month, item_name, total_quantity_sold)
                VALUES (source.sales_year, source.sales_month, source.item_name, source.total_quantity_sold);
        """)
    print(f"Aggregated sales data incrementally updated in Gold layer at: {gold_sales_path}")
    print("Current Gold Table Sample Data:")
    spark.sql("SELECT * FROM monthly_item_sales ORDER BY sales_year, sales_month, item_name").show()
