# --- Bronze Layer: Raw Data Ingestion ---
-- Bronze: Raw Customer Orders
-- Uses cloud_files to incrementally ingest JSON files.

COMMENT "Raw customer orders, as-is from source files. Uses Auto Loader for incremental ingestion."

  )

-- Bronze: Raw Customer Orders (using hardcoded path for demonstration)

-- --- Silver Layer: Cleansed & Conformed Data ---

-- Silver: Conformed Customer Orders (Fact Table)
CREATE STREAMING LIVE TABLE conformed_customer_orders
COMMENT "Cleaned and conformed customer orders (fact table) with data quality checks."
TBLPROPERTIES ("quality" = "silver")
-- Apply data quality expectations directly here.
-- DLT automatically handles dropping/quarantining based on pipeline config.
APPLY (CONSTRAINT valid_order_id EXPECT (OrderID IS NOT NULL AND TRIM(OrderID) <> '') ON VIOLATION DROP ROW)
APPLY (CONSTRAINT positive_quantity EXPECT (CAST(Quantity AS INT) > 0) ON VIOLATION DROP ROW)
AS SELECT
  CAST(OrderID AS INT) AS order_id,
  COALESCE(CustomerID, 'UNKNOWN') AS customer_id,
  Item AS item_name,
  CAST(Quantity AS INT) AS quantity,
  TO_DATE(OrderDate, 'yyyy-MM-dd') AS order_date, -- Handles 'yyyy/MM/dd' if TO_DATE supports it implicitly
  TO_TIMESTAMP(TimestampUTC, "yyyy-MM-dd HH:mm:ss'Z'") AS order_timestamp_utc,
  current_timestamp() AS processing_timestamp_silver
FROM STREAM(live.raw_customer_orders); -- Read incrementally from the Bronze table


-- Silver: Customer Dimension (SCD Type 2)
-- This table is managed by the APPLY CHANGES INTO statement for SCD Type 2 logic.
CREATE OR REFRESH STREAMING LIVE TABLE dim_customer
COMMENT "Customer Dimension Table (SCD Type 2) - Tracks history of customer attributes."
TBLPROPERTIES ("quality" = "silver");

-- Implement SCD Type 2 logic using APPLY CHANGES INTO
APPLY CHANGES INTO live.dim_customer
FROM STREAM(live.raw_customer_updates) -- Read incremental updates from Bronze
KEYS (CustomerID) -- Natural key for the dimension
SEQUENCE BY UpdateTimestamp -- Column to determine the order of changes (e.g., CDC timestamp)
STORED AS SCD TYPE 2 -- Specifies SCD Type 2 implementation
-- Optional: Columns that trigger a new version. If not specified, all non-key columns trigger.
-- track_history_for_columns = (CustomerName, CustomerEmail)
-- Optional: apply_as_deletes WHEN Operation = 'DELETE' -- If your source has explicit delete flags
;


-- --- Gold Layer: Curated & Business-Ready Data ---

-- Gold: Monthly Item Sales (Aggregate Table)
-- This is a MATERIALIZED VIEW (LIVE TABLE) in DLT. It reads the latest state of Silver.
-- DLT optimizes this to be incremental where possible, even though it's a batch aggregation.
CREATE LIVE TABLE monthly_item_sales
COMMENT "Monthly aggregated sales quantity per item, optimized for reporting."
TBLPROPERTIES ("quality" = "gold")
AS SELECT
  YEAR(order_date) AS sales_year,
  MONTH(order_date) AS sales_month,
  item_name,
  SUM(quantity) AS total_quantity_sold
FROM live.conformed_customer_orders -- Reads the LATEST state of the Silver table
GROUP BY 1, 2, 3;
