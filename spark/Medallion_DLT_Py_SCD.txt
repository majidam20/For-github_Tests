# --- Configuration for Data Paths ---

# Define raw schemas for initial data generation

    StructField("Item", StringType(), True),
    StructField("Quantity", StringType(), True),
    StructField("OrderDate", StringType(), True),

raw_customer_schema_for_gen = StructType([
    StructField("CustomerID", StringType(), True),
    StructField("CustomerName", StringType(), True),
    StructField("CustomerEmail", StringType(), True),
    StructField("UpdateTimestamp", StringType(), True), # Acts as sequence_by column
    StructField("Operation", StringType(), True) # For simulating deletes if needed, though SCD2 primarily handles updates/inserts
])

def create_initial_raw_data():

    Creates initial raw data files in the specified paths.

    but run as a separate setup script or be fed by external sources.
    """

    # Clean up previous data directories
    dbutils.fs.rm(raw_data_path_orders, True)
    dbutils.fs.rm(raw_data_path_customers, True)

        ("1001", "C001", "Laptop", "1", "2023-01-15", "2023-01-15 10:30:00Z"),

    ]

    print("Initial raw data files created.")

# --- Bronze Layer: Raw Data Ingestion ---
# These are 'streaming tables' because they use Auto Loader to continuously ingest new files.

@dlt.table(
    comment="Raw customer orders, as-is from source files. Uses Auto Loader for incremental ingestion.",
    table_properties={"quality": "bronze"}
)
def raw_customer_orders():
  return (
    spark.readStream.format("cloudFiles")
      .option("cloudFiles.format", "json") # Or "csv", "parquet" etc.
      .option("cloudFiles.schemaLocation", f"{raw_data_path_orders}/_schema") # DLT manages schema for Auto Loader
      .option("cloudFiles.inferSchema", "true") # Infer schema at Bronze for flexibility
      .load(raw_data_path_orders)
      .withColumn("ingestion_timestamp", current_timestamp())
  )

@dlt.table(
    comment="Raw customer updates for dimension, as-is. Used for SCD Type 2.",
    table_properties={"quality": "bronze"}
)
def raw_customer_updates():
  return (
    spark.readStream.format("cloudFiles")
      .option("cloudFiles.format", "json")
      .option("cloudFiles.schemaLocation", f"{raw_data_path_customers}/_schema") # DLT manages schema for Auto Loader
      .option("cloudFiles.inferSchema", "true") # Infer schema at Bronze for flexibility
      .load(raw_data_path_customers)
      .withColumn("ingestion_timestamp", current_timestamp())
  )


# --- Silver Layer: Cleansed & Conformed Data ---

@dlt.table(
    comment="Cleaned and conformed customer orders with standardized formats and data quality checks.",
    table_properties={"quality": "silver"}
)
@dlt.expect("valid_order_id", "OrderID IS NOT NULL AND TRIM(OrderID) <> ''")
@dlt.expect("positive_quantity", "CAST(Quantity AS INT) > 0")
def conformed_customer_orders():
  return (
    dlt.read_stream("raw_customer_orders") # Reads incrementally from the Bronze table
      .select(
          col("OrderID").cast(IntegerType()).alias("order_id"),
          coalesce(col("CustomerID"), lit("UNKNOWN")).alias("customer_id"),
          col("Item").alias("item_name"),
          col("Quantity").cast(IntegerType()).alias("quantity"),
          to_date(col("OrderDate"), "yyyy-MM-dd").alias("order_date"), # Standardize date format
          to_timestamp(col("TimestampUTC"), "yyyy-MM-dd HH:mm:ss'Z'").alias("order_timestamp_utc"),
          current_timestamp().alias("processing_timestamp_silver")
      )
  )

@dlt.create_streaming_table(

# --- Gold Layer: Curated & Business-Ready Data ---

@dlt.table(
    comment="Monthly aggregated sales quantity per item, optimized for reporting.",
    table_properties={"quality": "gold"}
)
def monthly_item_sales():
  return (
    dlt.read("conformed_customer_orders") # Reads the LATEST state of the Silver table (batch read)
      .groupBy(
          year(col("order_date")).alias("sales_year"),
          month(col("order_date")).alias("sales_month"),
          col("item_name")
      )
      .agg(
          sum("quantity").alias("total_quantity_sold")
      )
      .orderBy("sales_year", "sales_month", "item_name")
  )
